# 容灾配置

## 操作场景

用于配置DC内负荷分担，跨DC UPF成对主备容灾场景下SMF与UPF的配置。

## 必备事项

前提条件

* SMF与UPF已完成基础配置。
* 主备UPF间已完成静态地址路由冗余配置。

## 任务示例

任务描述

* DC1和DC2容灾，其中DC1内部署UPF1和UPF2，DC2内部署UPF3和UPF4，UPF1与UPF3主备容灾，UPF2与UPF4主备容灾。
* 终端静态签约地址，UPF发布网段路由。

脚本

1. //SMF配置

   ```
   ADD APN: APN="APN1";
   ADD NSDNN: NSIDX=0, DNN="APN1";
   SET APNPCCFUNC: APN="APN1", HOMEPCCSWITCH=DISABLE, ROAMPCCSWITCH=DISABLE, VISITPCCSWITCH=DISABLE;
   ADD APNUSRPROFG: APN="APN1", USERPROFGNAME="up_group_cmcc_cmnet";
   SET APNADDRESSATTR:ALLOCWITHOUTPCO=ENABLE, ANTISPOOFINGDI=ENABLE, ANTISPOOFINGUI=ENABLE, FRAMEDROUTE=DISABLE, IGNORECONFLICT=ENABLE, IPV4ALLOCTYPE=RADIUS, IPV4RADIUSPRIOR=ENABLE, IPV6ALLOCTYPE=RADIUS, IPV6RADIUSPRIOR=ENABLE, IPV6RAINTV=14400, IPV6RAOFLAG=ENABLE, RADIUSIPROUTE=ENABLE, DHCPIPROUTE=ENABLE, STATICIP=ENABLE, STATICIPROUTE=ENABLE, SUPPORTIPV4=ENABLE, SUPPORTIPV6=ENABLE, SUPPORTPRIOR=IPV4, APN="APN1", IPTYPEFORDUALIP=IPV4V6, SINGLEADDRCAUSE=NETWORK_PREFER, IPV4ALLOCPRIALG=DISABLE, IPV6ALLOCPRIALG=DISABLE, RDSSVRCTRL=ENABLE, IPV6RAMTUSW=DISABLE, IPV6MTU=1800, IPV6RALIFETIME=65535, IPV6RATIMES=0;

   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_1", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf01.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.10.2", PORT=8805;
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_2", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf02.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.20.2", PORT=8805;
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_3", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf03.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.11.2", PORT=8805;
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_4", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf04.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.21.2", PORT=8805;

   ADD PNFNS: INDEX=1, NFINSTANCEID="UPF_Instance_1", SST=1;
   ADD PNFNS: INDEX=2, NFINSTANCEID="UPF_Instance_2", SST=1;
   ADD PNFNS: INDEX=3, NFINSTANCEID="UPF_Instance_3", SST=1;
   ADD PNFNS: INDEX=4, NFINSTANCEID="UPF_Instance_4", SST=1;
   ADD PNFDNN: INDEX=1, NFINSTANCEID="UPF_Instance_1", DNN="APN1", PNFNSINDEX=1, PRISWITCH=INHERIT, CAPSWITCH=INHERIT;
   ADD PNFDNN: INDEX=2, NFINSTANCEID="UPF_Instance_2", DNN="APN1", PNFNSINDEX=2, PRISWITCH=INHERIT, CAPSWITCH=INHERIT;
   ```
