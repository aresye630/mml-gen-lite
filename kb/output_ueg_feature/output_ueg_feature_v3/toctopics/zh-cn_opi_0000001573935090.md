# 容灾配置

## 操作场景

用于配置DC内负荷分担，DC间整体主备容灾场景下SMF与UPF的配置。

此场景下UPF无需特殊配置。

## 必备事项

前提条件

SMF与UPF已完成基础配置。

## 任务示例

任务描述

* DC1和DC2容灾，其中DC1内部署UPF1和UPF2，DC2内部署UPF3和UPF4，DC内服务分担容灾，DC1与DC2主备容灾。
* 终端静态签约地址，UPF发布主机路由。

脚本

1. （SMF配置）配置APN及相关信息。

   ```
   ADD APN: APN="APN1";
   ADD NSDNN: NSIDX=0, DNN="APN1";
   SET APNPCCFUNC: APN="APN1, HOMEPCCSWITCH=DISABLE, ROAMPCCSWITCH=DISABLE, VISITPCCSWITCH=DISABLE;
   ADD APNUSRPROFG: APN="APN1", USERPROFGNAME="up_group_cmcc_cmnet";
   SET APNADDRESSATTR:ALLOCWITHOUTPCO=ENABLE, ANTISPOOFINGDI=ENABLE, ANTISPOOFINGUI=ENABLE, FRAMEDROUTE=DISABLE, IGNORECONFLICT=ENABLE, IPV4ALLOCTYPE=RADIUS, IPV4RADIUSPRIOR=ENABLE, IPV6ALLOCTYPE=RADIUS, IPV6RADIUSPRIOR=ENABLE, IPV6RAINTV=14400, IPV6RAOFLAG=ENABLE, RADIUSIPROUTE=ENABLE, DHCPIPROUTE=ENABLE, STATICIP=ENABLE, STATICIPROUTE=ENABLE, SUPPORTIPV4=ENABLE, SUPPORTIPV6=ENABLE, SUPPORTPRIOR=IPV4, APN="APN1", IPTYPEFORDUALIP=IPV4V6, SINGLEADDRCAUSE=NETWORK_PREFER, IPV4ALLOCPRIALG=DISABLE, IPV6ALLOCPRIALG=DISABLE, RDSSVRCTRL=ENABLE, IPV6RAMTUSW=DISABLE, IPV6MTU=1800, IPV6RALIFETIME=65535, IPV6RATIMES=0;
   ```
2. （SMF配置）配置用户面节点实例。配置Profile实例。

   ```
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_1", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf01.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.10.2", PORT=8805;
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_2", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf02.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.20.2", PORT=8805;
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_3", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf03.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.11.2", PORT=8805;
   ADD PNFPROFILE: NFINSTANCEID="UPF_Instance_4", NFTYPE=NfUPF, NFSTATUS=Registered, FQDN="upf04.node", IPADDRESSTYPE=IPTypeV4, IPV4ADDRESS1="192.168.21.2", PORT=8805;
   ```
3. （SMF配置）配置用户面节点的位置特征和分流能力。

   ```
   ADD UPNODE: NFINSTANCENAME="UPF_Instance_1", LOCATION=Central, UPFUNCTION=None, LOCK=FALSE, APSAMIGFUNC=DISABLE, ADDRALLOCMODE=INHERIT, SHAREDUPFSW=DISABLE, VPNNAME="VRF_CP", SUPATSSS=Disable, IPV6NDPLCY=PRIVATE, IOSWITCH=DISABLE, CAMPUSSW=DISABLE, ULCLFWDTNLSW=DISABLE, PDUESTWITHOUTN9=DISABLE, NGLANSW=DISABLE;
   ADD UPNODE: NFINSTANCENAME="UPF_Instance_2", LOCATION=Central, UPFUNCTION=None, LOCK=FALSE, APSAMIGFUNC=DISABLE, ADDRALLOCMODE=INHERIT, SHAREDUPFSW=DISABLE, VPNNAME="VRF_CP", SUPATSSS=Disable, IPV6NDPLCY=PRIVATE, IOSWITCH=DISABLE, CAMPUSSW=DISABLE, ULCLFWDTNLSW=DISABLE, PDUESTWITHOUTN9=DISABLE, NGLANSW=DISABLE;
   ADD UPNODE: NFINSTANCENAME="UPF_Instance_3", LOCATION=Central, UPFUNCTION=None, LOCK=FALSE, APSAMIGFUNC=DISABLE, ADDRALLOCMODE=INHERIT, SHAREDUPFSW=DISABLE, VPNNAME="VRF_CP", SUPATSSS=Disable, IPV6NDPLCY=PRIVATE, IOSWITCH=DISABLE, CAMPUSSW=DISABLE, ULCLFWDTNLSW=DISABLE, PDUESTWITHOUTN9=DISABLE, NGLANSW=DISABLE;
   ADD UPNODE: NFINSTANCENAME="UPF_Instance_4", LOCATION=Central, UPFUNCTION=None, LOCK=FALSE, APSAMIGFUNC=DISABLE, ADDRALLOCMODE=INHERIT, SHAREDUPFSW=DISABLE, VPNNAME="VRF_CP", SUPATSSS=Disable, IPV6NDPLCY=PRIVATE, IOSWITCH=DISABLE, CAMPUSSW=DISABLE, ULCLFWDTNLSW=DISABLE, PDUESTWITHOUTN9=DISABLE, NGLANSW=DISABLE;
   ```
4. （SMF配置）配置用户面节点支持的切片与DNN信息。

   ```
   ADD PNFNS: INDEX=1, NFINSTANCEID="UPF_Instance_1", SST=1;
   ADD PNFNS: INDEX=2, NFINSTANCEID="UPF_Instance_2", SST=1;
   ADD PNFNS: INDEX=3, NFINSTANCEID="UPF_Instance_3", SST=1;
   ADD PNFNS: INDEX=4, NFINSTANCEID="UPF_Instance_4", SST=1;
   ADD PNFDNN: INDEX=1, NFINSTANCEID="UPF_Instance_1", DNN="APN1", PNFNSINDEX=1, PRISWITCH=SPECIFIC, PRIORITY=1, CAPSWITCH=INHERIT;
   ADD PNFDNN: INDEX=2, NFINSTANCEID="UPF_Instance_2", DNN="APN1", PNFNSINDEX=2, PRISWITCH=SPECIFIC, PRIORITY=1, CAPSWITCH=INHERIT;
   ADD PNFDNN: INDEX=3, NFINSTANCEID="UPF_Instance_3", DNN="APN1", PNFNSINDEX=3, PRISWITCH=SPECIFIC, PRIORITY=100, CAPSWITCH=INHERIT;
   ADD PNFDNN: INDEX=4, NFINSTANCEID="UPF_Instance_4", DNN="APN1", PNFNSINDEX=4, PRISWITCH=SPECIFIC, PRIORITY=100, CAPSWITCH=INHERIT;
   ```
5. （SMF配置）配置APN粒度的优选UPF的开关。

   ```
   SET APNUPSELPLY: APN="APN1", PRIORITYFLAG=ENABLE, COMBINEDSELSTG=APNPRI;
   ```
6. （SMF配置）配置用户面节点的位置区信息，将位置区与TAI绑定。

   ```
   ADD UPAREA: AREANAME="UPAREA1", AREATYPE=N2TAI;
   ADD UPAREABINDN2TAI:AREANAME="UPAREA1", N2BEGINTAI="46011000BB8", N2ENDTAI="46011000BB8";
   ```
7. （SMF配置）配置用户面节点支持的为SMF提供服务区域的信息。

   ```
   ADD PNFSMFSERAREA:NFINSTANCEID="UPF_Instance_1", SMFSERVINGAREA="UPAREA1";
   ADD PNFSMFSERAREA:NFINSTANCEID="UPF_Instance_2", SMFSERVINGAREA="UPAREA1";
   ADD PNFSMFSERAREA:NFINSTANCEID="UPF_Instance_3", SMFSERVINGAREA="UPAREA1";
   ADD PNFSMFSERAREA:NFINSTANCEID="UPF_Instance_4", SMFSERVINGAREA="UPAREA1";
   ```
