# 配置PCF基础数据

## 操作场景

本场景用于在PCF上配置用户的签约数据。

## 对系统的影响

无。

## 任务示例

任务描述

配置终端开户数据和签约数据。

脚本

1. 配置终端开户数据。

   ```
   ADD PSUB: USRIDENTIFIER="460020000000001", USRMSISDN="8614xxxxx631335", USRSTATE=Normal, USRPAIDTYPE=NULL, USRSTATION=Master, USRCONTACTMETHOD=SMS, USRBILLCYCLEDATE=NULL, USRSUBNETTYPE=MBB, USRLANGUAGE=NULL, USRSMSRECEIVEFLAG=All, USRVERSTATUS=Normal;
   ```
2. 配置终端签约数据。

   ```
   ADD PSRV: USRIDENTIFIER="460020000000001", SRVNAME="9cu00_03_ffffff_testqci4";
   ```
3. 配置URSP策略数据。

   ```
   /*添加动作*/
   ADD NGACTION:ACTIONNAME="AG_ConneCapab",TYPE="ConnectionCapabilities",DESCRIPTION="ConnectionCapabilities01";
   ADD NGACTION:ACTIONNAME="AG_TrafficDes",TYPE="TrafficDescriptor",DESCRIPTION="TrafficDescriptor01";
   ADD NGACTION:ACTIONNAME="AG_RouteSelectDes",TYPE="RouteSelectionDescriptor",DESCRIPTION="RouteSelectionDescriptor01";
   ADD NGACTION:ACTIONNAME="AG_UrspRule",TYPE="UrspRule",DESCRIPTION="UrspRule01";
   ADD NGACTION:ACTIONNAME="AG_UepolicyPart",TYPE="UepolicyPart",DESCRIPTION="UepolicyPart01";
   ADD NGACTION:ACTIONNAME="AG_Instruction",TYPE="Instruction",DESCRIPTION="Instruction01";
   ADD NGACTION:ACTIONNAME="AG_UePolicySublist",TYPE="UePolicySublist",DESCRIPTION="UePolicySublist01";
   ADD NGACTION:ACTIONNAME="AG_UePolicyAction",TYPE="UePolicyAction",DESCRIPTION="5GActionGroup01";

   /*添加动作属性*/
   ADD NGACTIONATTR:ACTIONNAME="AG_ConneCapab",ATTRNAME="ims",TYPE=Value,VALUE="True";
   ADD NGACTIONATTR:ACTIONNAME="AG_ConneCapab",ATTRNAME="internet",TYPE=Value,VALUE="True";
   ADD NGACTIONATTR:ACTIONNAME="AG_ConneCapab",ATTRNAME="supl",TYPE=Value,VALUE="True";
   ADD NGACTIONATTR:ACTIONNAME="AG_ConneCapab",ATTRNAME="mms",TYPE=Value,VALUE="True";
   ADD NGACTIONATTR:ACTIONNAME="AG_TrafficDes",ATTRNAME="matchAll",TYPE=Value,VALUE="Yes";
   ADD NGACTIONATTR:ACTIONNAME="AG_TrafficDes",ATTRNAME="connectionCapabilities",TYPE=Reference ActionGroup,VALUE="AG_ConneCapab";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="sscMode",TYPE=Value,VALUE="SSC_MODE_1";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="precedence",TYPE=Value,VALUE="0";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="dnn",TYPE=Value,VALUE="huawei.com";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="snssai",TYPE=Value,VALUE="0100000a";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="preferredAccessType",TYPE=Value,VALUE="3GPP_ACCESS";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="offloadIndication",TYPE=Value,VALUE="False";
   ADD NGACTIONATTR:ACTIONNAME="AG_RouteSelectDes",ATTRNAME="pduSessionType",TYPE=Value,VALUE="IPV4V6";
   ADD NGACTIONATTR:ACTIONNAME="AG_UrspRule",ATTRNAME="trafficDescriptor",TYPE=Reference ActionGroup,VALUE="AG_TrafficDes";
   ADD NGACTIONATTR:ACTIONNAME="AG_UrspRule",ATTRNAME="precedence",TYPE=Value,VALUE="0";
   ADD NGACTIONATTR:ACTIONNAME="AG_UrspRule",ATTRNAME="routeSelectionDescriptor",TYPE=Reference ActionGroup,VALUE="AG_RouteSelectDes";
   ADD NGACTIONATTR:ACTIONNAME="AG_UepolicyPart",ATTRNAME="urspRule",TYPE=Reference ActionGroup,VALUE="AG_UrspRule";
   ADD NGACTIONATTR:ACTIONNAME="AG_Instruction",ATTRNAME="uepolicyPart",TYPE=Reference ActionGroup,VALUE="AG_UepolicyPart";
   ADD NGACTIONATTR:ACTIONNAME="AG_Instruction",ATTRNAME="upsc",TYPE=Value,VALUE="100";
   ADD NGACTIONATTR:ACTIONNAME="AG_UePolicySublist",ATTRNAME="mcc",TYPE=Value,VALUE="460";
   ADD NGACTIONATTR:ACTIONNAME="AG_UePolicySublist",ATTRNAME="mnc",TYPE=Value,VALUE="06";
   ADD NGACTIONATTR:ACTIONNAME="AG_UePolicySublist",ATTRNAME="instruction",TYPE=Reference ActionGroup,VALUE="AG_Instruction";
   ADD NGACTIONATTR:ACTIONNAME="AG_UePolicyAction",ATTRNAME="uePolicySublist",TYPE=Reference ActionGroup,VALUE="AG_UePolicySublist";
   ADD NGACTIONATTR:ACTIONNAME="AG_UePolicyAction",ATTRNAME="suppFeat",TYPE=Value,VALUE="9999999";

   /*添加规则*/
   ADD RULE:RULENAME="Rule_UEPolicy",TYPE=Amf Pcc Rule 5G,RELATION=AND;

   /*添加规则动作*/
   ADD RULENGACTION:RULENAME="Rule_UEPolicy",ACTIONNAME="AG_UePolicyAction";

   /*添加策略*/
   ADD POLICY:POLICYNAME="Policy_UE",TYPE=N15 Policy;

   /*添加策略规则*/
   ADD POLICYRULE:POLICYNAME="Policy_UE",RULENAME="Rule_UEPolicy";

   /*添加策略触发器*/
   ADD POLICYTRIGGER:POLICYNAME="Policy_UE",TRIGGERNAME="N15UserLocationChange";
   ADD POLICYTRIGGER:POLICYNAME="Policy_UE",TRIGGERNAME="N15PresenceReportingAreaChange";
   ADD POLICYTRIGGER:POLICYNAME="Policy_UE",TRIGGERNAME="N15ServAreaChange";
   ADD POLICYTRIGGER:POLICYNAME="Policy_UE",TRIGGERNAME="N15RfspChange";
   ADD POLICYTRIGGER:POLICYNAME="Policy_UE",TRIGGERNAME="N15PolicyCreateRequest";

   /*添加业务*/
   ADD SERVICE:SERVICENAME="Service_UE",SERVICETYPE=VALUE_ADDED_SERVICE,ACTIVATEDBY=PCEF,PRECEDENCE=0,QOSMODE=Replace,SUBSCRIPTIONFORCED=Yes,PROBILLDATE=No,QUOTAALLOCATIONMODE=None,ISSETSHAREDMODE=No,RESUBSCRIPTIONOPERATION=NULL,ISGROUPSERVICE=No,ISTEMPLATE=No,ISRECUSRLOC=No,APNMATCHTYPE=Full Matching,HASBASESERVICE=No,ADDONLEVEL=Exhaust,CALCVALIDFORDYNSUBGROUP=No;

   /*添加业务策略
   ADD SERVICEPOLICY:SERVICENAME="Service_UE",POLICYNAME="Policy_UE";
   ```
