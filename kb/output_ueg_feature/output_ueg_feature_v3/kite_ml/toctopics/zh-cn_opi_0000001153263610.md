## 操作场景

费率标识可以是费率组（Rating Group，RG）；也可以是费率组（Rating Group，RG）+业务ID（Service ID，SID）即（RG，SID）。使用（RG，SID）做费率标识可以进行更细粒度的配额管理。SMF支持内容计费，即按用户使用的业务粒度进行计费。对应内容计费的不同的业务，对应的费率标识不同。

适用于SMF。

## 必备事项

前提条件

* 请详细阅读融合计费原理。
* 使用内容计费时，已申请内容计费的License项。
* 完成加载License。

## 操作步骤

* 配置内容计费费率标识。
  1. 配置内容计费使用的URR组及相应URR。

     1. 配置URR。

        **ADD CPURR**
     2. 配置URR组，将URR绑定到URR组上。

        **ADD CPURRGROUP**
  2. 配置内容计费使用的Rule，将URR组绑定到对应的Rule上。

     1. 配置PCC策略组，将URR组绑定到PCC策略组。

        **ADD CPPCCPOLICYGRP**
     2. 配置Rule，将PCC策略组绑定到Rule上。

        **ADD CPRULE**
  3. 配置UserProfile，将Rule绑定到UserProfile上。

     1. 配置UserProfile。

        **ADD CPUSERPROFILE**
     2. 绑定该UserProfile的缺省URR组，建议每个UserProfile都绑定缺省费率URR组。

        **SET CPURRGRPBINDING**
     3. 将Rule绑定到UserProfile上。

        **ADD CPRULEBINDING**
  4. 配置UserProfile组及DNN实例。

     1. 配置UserProfile组，将UserProfile绑定到UserProfile组。

        **ADD USRPROFGROUP**

        **ADD UPBINDUPG**
     2. 配置DNN实例，将UserProifle组绑定到DNN实例。

        **ADD CPAPN**

        **ADD APNUSRPROFG**

        每个DNN实例只能绑定一个UserProfile组。如UserProfile组和DNN实例已配置且已绑定，本步骤只需执行**ADD UPBINDUPG**命令，将UserProfile绑定到UserProfile组即可。

脚本

//以下内容计费配置举例和UPF上的内容计费配置匹配。

//配置URR。URRID、USAGERPTMODE、OFFMETERINGTYPE、ONLMETERINGTYPE在SMF和UPF上的配置需要保持一致。

```
ADD CPURR: URRNAME="urr_online_URL", URRID=2300, USAGERPTMODE=ONLINE, ONLCOMPOUNDTYPE=ONLYRG, ONLINERG=100, ONLMETERINGTYPE=VOLUME;
ADD CPURR: URRNAME="urr_offline_URL", URRID=12500, USAGERPTMODE=OFFLINE, OFFCOMPOUNDTYPE=ONLYRG, RG=1100, OFFMETERINGTYPE=VOLUME;
ADD CPURR: URRNAME="urr_online_IMS", URRID=3300, USAGERPTMODE=ONLINE, ONLCOMPOUNDTYPE=ONLYRG, ONLINERG=200, ONLMETERINGTYPE=VOLUME;
ADD CPURR: URRNAME="urr_offline_IMS", URRID=13500, USAGERPTMODE=OFFLINE, OFFCOMPOUNDTYPE=ONLYRG, RG=1200, OFFMETERINGTYPE=VOLUME;
ADD CPURR: URRNAME="urr_online_any", URRID=1300, USAGERPTMODE=ONLINE, ONLCOMPOUNDTYPE=ONLYRG, ONLINERG=300, ONLMETERINGTYPE=VOLUME;
ADD CPURR: URRNAME="urr_offline_any", URRID=11500, USAGERPTMODE=OFFLINE, OFFCOMPOUNDTYPE=ONLYRG, RG=1300, OFFMETERINGTYPE=VOLUME;
ADD CPURR: URRNAME="URR_abnormal_online", URRID=4300, USAGERPTMODE=ONLINE, ONLCOMPOUNDTYPE=ONLYRG, ONLINERG=400, ONLMETERINGTYPE=FREE;
ADD CPURR: URRNAME="URR_abnormal_offline", URRID=14500, USAGERPTMODE=OFFLINE, OFFCOMPOUNDTYPE=ONLYRG, RG=1400, OFFMETERINGTYPE=FREE;
```

/配置URRGROUP。

```
ADD CPURRGROUP: URRGROUPNAME="UrrGp_URL", UPURRNAME1="urr_online_URL", DOWNURRNAME1="urr_online_URL", UPURRNAME2="urr_offline_URL", DOWNURRNAME2="urr_offline_URL";
ADD CPURRGROUP: URRGROUPNAME="UrrGp_IMS", UPURRNAME1="urr_online_IMS", DOWNURRNAME1="urr_online_IMS", UPURRNAME2="urr_offline_IMS", DOWNURRNAME2="urr_offline_IMS";
ADD CPURRGROUP: URRGROUPNAME="UrrGp_any", UPURRNAME1="urr_online_any", DOWNURRNAME1="urr_online_any", UPURRNAME2="urr_offline_any", DOWNURRNAME2="urr_offline_any";
ADD CPURRGROUP: URRGROUPNAME="UrrGp_abnormal", UPURRNAME1="URR_abnormal_online", DOWNURRNAME1="URR_abnormal_online", UPURRNAME2="URR_abnormal_offline", DOWNURRNAME2="URR_abnormal_offline";
```

//配置PCC组策略数据。

```
ADD CPPCCPOLICYGRP: PCCPOLICYGRPNM="pccpolicy_URL", URRGROUPNAME="UrrGp_URL";
ADD CPPCCPOLICYGRP: PCCPOLICYGRPNM="pccpolicy_IMS", URRGROUPNAME="UrrGp_IMS";
ADD CPPCCPOLICYGRP: PCCPOLICYGRPNM="pccpolicy_any", URRGROUPNAME="UrrGp_any";
```

//配置规则，并绑定已配置的PCC策略组。SMF、UPF上RULENAME以及RULE下配置的POLICYTYPE、POLICYNAME需要保持一致。

```
ADD CPRULE: RULENAME="Rule001_dns", POLICYTYPE=PCC, PRIORITY=100, POLICYNAME="pccpolicy_URL";
ADD CPRULE: RULENAME="Rule002_IMS", POLICYTYPE=PCC, PRIORITY=110, POLICYNAME="pccpolicy_IMS";
ADD CPRULE: RULENAME="Rule003_any", POLICYTYPE=PCC, PRIORITY=65000, POLICYNAME="pccpolicy_any";
ADD CPRULE: RULENAME="Rule003_any_IPv6", POLICYTYPE=PCC, PRIORITY=65010, POLICYNAME="pccpolicy_any";
```

//配置UserProfile，将Rule绑定到UserProfile上。USERPROFILE的名称统一规划，保证在PCF、SMF、UPF间名称一致。不一致时PCF下发的策略无法生效。

```
ADD CPUSERPROFILE: USERPROFILENAME="userprofile001", UPTYPE=PCC;
ADD CPUSERPROFILE: USERPROFILENAME="userprofile002", UPTYPE=PCC;
ADD CPRULEBINDING: USERPROFILENAME="userprofile001", RULENAME="Rule001_dns";
ADD CPRULEBINDING: USERPROFILENAME="userprofile001", RULENAME="Rule003_any";
ADD CPRULEBINDING: USERPROFILENAME="userprofile001", RULENAME="Rule003_any_IPv6";
ADD CPRULEBINDING: USERPROFILENAME="userprofile002", RULENAME="Rule002_IMS";
```

//配置规则组下的URR组。建议每个UserProfile都绑定缺省费率URR组。

```
SET CPURRGRPBINDING:USERPROFILENAME="userprofile001",DFTURRGRPNAME="UrrGp_abnormal";
SET CPURRGRPBINDING:USERPROFILENAME="userprofile002",DFTURRGRPNAME="UrrGp_abnormal";
```

//配置UserProfile组，将UserProfile绑定到UserProfile组。

```
ADD USRPROFGROUP: USERPROFGNAME="upg_test";
ADD UPBINDUPG: USERPROFGNAME="upg_test", UPBINDTYPE=SPECIFIC, PRIORITY=10, USERPROFILENAME="userprofile001";
ADD UPBINDUPG: USERPROFGNAME="upg_test", UPBINDTYPE=SPECIFIC, PRIORITY=20, USERPROFILENAME="userprofile002";
```

//配置DNN实例，将UserProfile组绑定到DNN实例。

```
ADD CPAPN: APN="apn-test", HASVPN=DISABLE, HASVPNIPV6=DISABLE, RESTORPGWSWITCH=INHERIT;
ADD APNUSRPROFG: APN="apn-test", USERPROFGNAME="upg_test";
```
