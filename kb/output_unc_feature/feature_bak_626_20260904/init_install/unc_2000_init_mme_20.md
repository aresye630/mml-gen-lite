# MME典型配置实例

- 操作场景
- 必备事项
- 操作步骤

## 操作场景

本任务实例实现以下场景配置：UNC通过IP骨干网与eNodeB、HSS、S-GW、P-GW、DNS互通。UNC到eNodeB、HSS和S-GW都采用了双平面+VRF+OSPF动态路由+BFD的组网方式，其中UNC到eNodeB采用SCTP非多归属，UNC到HSS采用SCTP双归属。

**图1** 典型组网图

## 必备事项

前提条件

- 请了解UNC各类接口的类型及命名规范，可参见业务组网与逻辑接口介绍。
- 基础数据配置、IP路由组网和业务接口配置子任务在前述章节已有详细描述，可参见基础数据配置、配置VNF侧IP路由数据（非SDN）和配置MME。本文针对典型的任务场景举例说明整网脚本配置。

数据

参考各子任务的数据规划准备。

文档

获取网设LLD文档。

## 操作步骤

1. 进入“MML命令行-UNC”窗口。
2. 配置本局信息，包括归属PLMN和MMEID。

   ```
   ADD HPLMN: MCC="123", MNC="001", CC="86", S5S8TYPE=GTP; 
   ADD MMEID: MCC="123", MNC="001", MMEGI="FF33", MMEC="01";
   SET SYS: CAP=255;
   ```
3. 配置S6a数据。
   1. 配置动态路由OSPF。

      1. 配置VPN实例。

         ```
         ADD L3VPNINST:VRFNAME="vpn_SIG",VRFDESCRIPTION="vpn_SIG"; 
         ADD VPNINSTAF:VRFNAME="vpn_SIG",AFTYPE=ipv4uni;
         ```
      2. 配置主接口的管理状态为up。

         ```
         MOD INTERFACE:IFNAME="Ethernet64/0/8",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=TRUE,IFTRAPENABLE=TRUE,IFSTATITVL=5; 
         MOD INTERFACE:IFNAME="Ethernet64/0/9",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=TRUE,IFTRAPENABLE=TRUE,IFSTATITVL=5; 
         MOD INTERFACE:IFNAME="Ethernet65/0/8",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=TRUE,IFTRAPENABLE=TRUE,IFSTATITVL=5;
         MOD INTERFACE:IFNAME="Ethernet65/0/9",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=TRUE,IFTRAPENABLE=TRUE,IFSTATITVL=5;
         ```
      3. 配置以太网接口。

         ```
         ADD INTERFACE:IFNAME="Ethernet64/0/8.2130",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=FALSE,IFTRAPENABLE=TRUE,IFSTATITVL=5; 
         ADD INTERFACE:IFNAME="Ethernet65/0/9.2131",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=FALSE,IFTRAPENABLE=TRUE,IFSTATITVL=5; 
         ADD INTERFACE:IFNAME="Ethernet64/0/9.2131",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=FALSE,IFTRAPENABLE=TRUE,IFSTATITVL=5; 
         ADD INTERFACE:IFNAME="Ethernet65/0/8.2130",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=FALSE,IFTRAPENABLE=TRUE,IFSTATITVL=5; 
         ADD INTERFACE:IFNAME="LoopBack130",IFADMINSTATUS=up,IFMTU=1500,IFDF=FALSE,IFSTATIENABLE=TRUE,IFTRAPENABLE=TRUE,IFSTATITVL=5;
         ```
      4. 增加接口绑定VPN。

         ```
         ADD IPBINDVPN:IFNAME="Ethernet64/0/8.2130",VRFNAME="vpn_SIG"; 
         ADD IPBINDVPN:IFNAME="Ethernet65/0/9.2131",VRFNAME="vpn_SIG"; 
         ADD IPBINDVPN:IFNAME="Ethernet64/0/9.2131",VRFNAME="vpn_SIG"; 
         ADD IPBINDVPN:IFNAME="Ethernet65/0/8.2130",VRFNAME="vpn_SIG"; 
         ADD IPBINDVPN:IFNAME="LoopBack130",VRFNAME="vpn_SIG";
         ```
      5. 在以太网子接口上关联VLAN ID。

         ```
         ADD ETHSUBIF:IFNAME="Ethernet64/0/8.2130",SUBIFTYPE=VlanType,VLANTYPEVID=2130; 
         ADD ETHSUBIF:IFNAME="Ethernet65/0/9.2131",SUBIFTYPE=VlanType,VLANTYPEVID=2131; 
         ADD ETHSUBIF:IFNAME="Ethernet64/0/9.2131",SUBIFTYPE=VlanType,VLANTYPEVID=2131; 
         ADD ETHSUBIF:IFNAME="Ethernet65/0/8.2130",SUBIFTYPE=VlanType,VLANTYPEVID=2130;
         ```
      6. 配置外联口的物理IP。

         ```
         ADD IFIPV4ADDRESS:IFNAME="Ethernet64/0/8.2130",IFIPADDR="10.220.82.98",SUBNETMASK="255.255.255.240",ADDRTYPE=main; 
         ADD IFIPV4ADDRESS:IFNAME="Ethernet65/0/9.2131",IFIPADDR="10.220.82.115",SUBNETMASK="255.255.255.240",ADDRTYPE=main; 
         ADD IFIPV4ADDRESS:IFNAME="Ethernet64/0/9.2131",IFIPADDR="10.220.82.114",SUBNETMASK="255.255.255.240",ADDRTYPE=main; 
         ADD IFIPV4ADDRESS:IFNAME="Ethernet65/0/8.2130",IFIPADDR="10.220.82.99",SUBNETMASK="255.255.255.240",ADDRTYPE=main; 
         ADD IFIPV4ADDRESS:IFNAME="LoopBack130",IFIPADDR="10.220.95.25",SUBNETMASK="255.255.255.255",ADDRTYPE=main;
         ```
      7. 创建OSPF进程。

         ```
         ADD OSPF:PROCID=668,VRFNAME="vpn_SIG",SCHEMAROUID="10.220.95.25",SILENTALLFLAG=FALSE,BFDALLINTFFLG=TRUE,BFDRXCFGFLAG=TRUE,BFDMINRXINTV=500,BFDTXCFGFLAG=TRUE,BFDMINTXINTV=500,SCHEMABFDDETMUL=8,RFC1583COMPFLG=TRUE,ROUTAGFLAG=FALSE,BANDWIDTHREF=100,RXMTLIMITENABLE=FALSE,OPAQCAPFLG=FALSE,LSAARRINTVFLAG=FALSE,LSAARRMAXINTV=1000,LSAARRSTARINTV=500,LSAARRHLDINTV=500,LSAORIGINSTFLAG=FALSE,LSAORIGINTV=5,LSAORIGMAXINTV=5000,LSAORIGSTARINTV=500,LSAORIGHLDINTV=1000,VPNINSCAPSIMFLG=TRUE,STUBROUFLG=NoConfig,SAFESYNCFLG=FALSE,RTTAGDISFLAG=FALSE,ECAROUTEFLAG=FALSE,DOMAINNULLFLAG=FALSE,SUPPREACH=FALSE,SPFCONSERVFLAG=FALSE,SHAMHELLOFLAG=FALSE,MPLSLDPAUTOCFG=None,VIRTUALSYSFLAG=TRUE;
         ```
      8. 增加OSPF区域。

         ```
         ADD OSPFAREA:PROCID=668,AREAID="10.0.0.33",AREATYPE=NSSA,NSSADEFAULTROUTEADVERTISE=FALSE,NSSANOIMPORTROUTE=FALSE,NSSANOSUMMARY=FALSE,NSSASETNBIT=FALSE,NSSASUPPRESSFORWARDINGADDRESS=FALSE,NSSATRANSLATORALWAYS=FALSE,NSSATRANSLATORINTERVAL=40,NSSAZEROADDRESSFORWARDING=FALSE,DEFAULTCOST=1;
         ```
      9. 增加运行OSPF协议的接口。

         ```
         ADD OSPFNETWORK:PROCID=668,AREAID="10.0.0.33",IPADDRESS="10.220.82.96",WILDCARDMASK="0.0.0.15"; 
         ADD OSPFNETWORK:PROCID=668,AREAID="10.0.0.33",IPADDRESS="10.220.82.112",WILDCARDMASK="0.0.0.15"; 
         ADD OSPFNETWORK:PROCID=668,AREAID="10.0.0.33",IPADDRESS="10.220.95.25",WILDCARDMASK="0.0.0.0";
         ```
      10. 设置OSPF引入路由类型。

          ```
          ADD OSPFIMPORTROUTE:PROCID=668,TOPOID=0,PROTOCOL=wlr,IMPTCOSTCFG=FALSE,IMPTTAGCFG=FALSE,IMPTTYPECFG=FALSE;
          ```
   2. 配置BFD。

      1. 全局激活BFD。

         ```
         SET BFD:BFDENABLE=TRUE,DELAYUPTIMER=0,TOSEXP=7;
         ```
      2. 以接口粒度配置动态BFD。

         ```
         ADD OSPFINTERFACE:PROCID=668,AREAID="10.0.0.33",IFNAME="Ethernet64/0/8.2130",DRPRI=0,NETWORKTYPE=broadcast,TIMERCONSERFLAG=FALSE,CONFIGCOST=1500,BFDENABLE=TRUE,MINRXRECVINTV=500,MINTXTRANINTV=500,DETECTMULINTV=4,VIRTUALSYSFLAG=TRUE; 
         ADD OSPFINTERFACE:PROCID=668,AREAID="10.0.0.33",IFNAME="Ethernet65/0/9.2131",DRPRI=0,NETWORKTYPE=broadcast,TIMERCONSERFLAG=FALSE,CONFIGCOST=1500,BFDENABLE=TRUE,MINRXRECVINTV=500,MINTXTRANINTV=500,DETECTMULINTV=4,VIRTUALSYSFLAG=TRUE; 
         ADD OSPFINTERFACE:PROCID=668,AREAID="10.0.0.33",IFNAME="Ethernet64/0/9.2131",DRPRI=0,NETWORKTYPE=broadcast,TIMERCONSERFLAG=FALSE,CONFIGCOST=1500,BFDENABLE=TRUE,MINRXRECVINTV=500,MINTXTRANINTV=500,DETECTMULINTV=4,VIRTUALSYSFLAG=TRUE; 
         ADD OSPFINTERFACE:PROCID=668,AREAID="10.0.0.33",IFNAME="Ethernet65/0/8.2130",DRPRI=0,NETWORKTYPE=broadcast,TIMERCONSERFLAG=FALSE,CONFIGCOST=1500,BFDENABLE=TRUE,MINRXRECVINTV=500,MINTXTRANINTV=500,DETECTMULINTV=4,VIRTUALSYSFLAG=TRUE;
         ```
   3. 配置S6a接口业务IP。

      ```
      ADD SERVICEIP:IPVERSION=IPv4,SERVICEIPV4="10.220.96.136",VPNINSTNAME="vpn_SIG",DESC="S6a"; 
      ADD SERVICEIP:IPVERSION=IPv4,SERVICEIPV4="10.220.96.137",VPNINSTNAME="vpn_SIG",DESC="S6a";
      ```
   4. 配置S6a接口基于IP的宽带信令SCTP偶联配置的SCTP协议参数。

      ```

   5. 配置Diameter链路数据。

      1. 配置Diameter本端信息。

         ```

      4. 配置Diameter链路信息。

         ADD DMLNK:VPNNAME="vpn_SIG",LINKIDX=2,IPTYPE=TPTADDR_TYPE_IPV4,PROTOTYPE=CONN_PROTOCOL_SCTP,LOCALIPV4_1="10.220.96.136",LOCALIPV4_2="10.220.96.137",LOCALPORT=8521,PEERIPV4_1="10.133.221.12",PEERIPV4_2="10.133.221.4",PEERPORT=3868,CLIORSER=DIAM_CONN_CLIENT,LINKSIDX=1,SCTPINDX=20,LINKNAM="TEHSSH01_C_02",CROSSIPFLAG=NO;

      5. 配置到HSS的主机路由。

         ```
         ADD DMHOSTRT:ROUTEIDX=1,APPNAM=S6A/S6D,RSELMODE=SELMODE_MASTER_SLAVE,PEERSEL=PEER_INDEX,PEERIDX=1,ROUTENAM="TEHSSH01_HR",PRIORITY=0;

         ```
