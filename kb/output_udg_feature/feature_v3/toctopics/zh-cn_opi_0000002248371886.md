## 操作场景

媒体中继业务中，UPF需要识别DNS请求中是否携带OTT规划的relay域名来作DNS代理，此时需要在UPF上配置对应的业务计费规则。

## 必备事项

前提条件

已参考增加媒体服务商基础配置~配置点播业务的安全认证策略完成媒体中继业务相关配置。

## 任务示例

任务描述

基于规划的点播域名**relay.spvod.com**，直播域名**relay.spliving.com**配置L34层、L7层识别规则，绑定到媒体中继签约套餐对应的预定义规则**testuserprofilename**。

脚本

//配置FlowFilter

```
ADD FLOWFILTER:FLOWFILTERNAME="sp_relay_00_f_f_h_001", TETHERDETFLAG=NONE, FLOWFILTERTYPE=USERCONFIG;
ADD FLOWFILTER:FLOWFILTERNAME="sp_relay_00_f_f_h_002", TETHERDETFLAG=NONE, FLOWFILTERTYPE=USERCONFIG;
ADD FLOWFILTER:FLOWFILTERNAME="sp_relay_00_f_f_h_003", TETHERDETFLAG=NONE, FLOWFILTERTYPE=USERCONFIG;
ADD FLOWFILTER:FLOWFILTERNAME="sp_relay_00_f_f_h_004", TETHERDETFLAG=NONE, FLOWFILTERTYPE=USERCONFIG;
```

//针对直播业务，配置L34 HOST规则绑定到FlowFilter，配置L7层匹配规则绑定到FlowFilter。

```
ADD HOST:HOSTNAME="sp_relay_live_001_h_100", DOMAINVALUE="relay.spliving.com", PRIORITY=33800;
ADD FILTER:FILTERNAME="sp_relay_live_001_f_100", L34PROTTYPE=STRING, L34PROTOCOL=ANY, SVRIPMODE=HOSTNAME, HOSTNAME="sp_relay_live_001_h_100", MSSTARTPORT=0, MSENDPORT=65535, SVRSTARTPORT=0, SVRENDPORT=65535;
ADD FILTERIPV6:FILTERNAME="ipv6_sp_relay_live_001_f_100", L34PROTTYPEV6=STRING, L34PROTOCOL=ANY, SVRIPMODE=HOSTNAME, HOSTNAME="sp_relay_live_001_h_100", MSSTARTPORT=0, MSENDPORT=65535, SVRSTARTPORT=0, SVRENDPORT=65535;
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_00_f_f_h_001", FILTERNAME="sp_relay_live_001_f_100";
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_00_f_f_h_001", FILTERNAME="ipv6_sp_relay_live_001_f_100";
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_00_f_f_h_002", FILTERNAME="sp_relay_live_001_f_100";
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_00_f_f_h_002", FILTERNAME="ipv6_sp_relay_live_001_f_100";

ADD L7FILTER:L7FILTERNAME="sp_relay_00_l7_f_001", SUBL7FLTNAME="sp_relay_00_sl7_f_100", URLTYPE=URL, URL="*relay.spliving.com/*", METHODTYPE=CONNECT-1&DELETE-1&GET-1&HEAD-1&OPTIONS-1&POST-1&PUT-1&TRACE-1, ISREFEREREN=DISABLE, MMSBLURMATCH=DISABLE;
ADD L7FILTER:L7FILTERNAME="sp_relay_00_l7_f_002", SUBL7FLTNAME="sp_relay_00_sl7_f_100", URLTYPE=URL, URL="*relay.spliving.com", METHODTYPE=CONNECT-1&DELETE-1&GET-1&HEAD-1&OPTIONS-1&POST-1&PUT-1&TRACE-1, ISREFEREREN=DISABLE, MMSBLURMATCH=DISABLE;
ADD PROTBINDFLOWF:FLOWFILTERNAME="sp_relay_00_f_f_h_001", PROTOCOLNAME="http", L7FILTERNAME="sp_relay_00_l7_f_001";
ADD PROTBINDFLOWF:FLOWFILTERNAME="sp_relay_00_f_f_h_002", PROTOCOLNAME="https", L7FILTERNAME="sp_relay_00_l7_f_002";
```

//针对点播业务，配置L34 HOST规则绑定到FlowFilter，配置L7层匹配规则绑定到FlowFilter。

//增加内容计费费率（未单独规划费率时，不需要新配置，复用原有费率）

如果需要增加计费费率配置时，需要同步在SMF上增加对应的URR费率等相关配置，避免出现SMF与UPF的计费配置不一致，触发告警。

```
ADD URR:URRNAME="sp_relay_00_cbb_offline", URRID=1750010088, USAGERPTMODE=OFFLINE, OFFMETERINGTYPE=FREE;
ADD URR:URRNAME="sp_relay_00_cbb_online", URRID=1750000088, USAGERPTMODE=ONLINE, ONLMETERINGTYPE=FREE;
ADD URRGROUP:URRGROUPNAME="sp_relay_00_ch_p", UPURRNAME1="sp_relay_00_cbb_offline", DOWNURRNAME1="sp_relay_00_cbb_offline", UPURRNAME2="sp_relay_00_cbb_online", DOWNURRNAME2="sp_relay_00_cbb_online";
ADD PCCPOLICYGRP:PCCPOLICYGRPNM="sp_relay_00_pcc_p_g", FUPSESSIONEXC=DISABLE, SIGNALASSOCIATE=ENABLE, ADCMUTEFLAG=ENABLE, HTTP2DEGRADESW=INHERIT, TOKENFUNCFLAG=DISABLE, URRGROUPNAME="sp_relay_00_ch_p";
```

//配置业务匹配规则并绑定到媒体中继签约套餐。

```
ADD RULE:RULENAME="sp_relay_00_r_h_001", POLICYTYPE=PCC, FLOWFILTERNAME="sp_relay_00_f_f_h_001", PRIORITY=30280, POLICYNAME="sp_relay_00_pcc_p_g", FILTERINGMODE=FLOWFILTER;
ADD RULE:RULENAME="sp_relay_00_r_h_002", POLICYTYPE=PCC, FLOWFILTERNAME="sp_relay_00_f_f_h_002", PRIORITY=30281, POLICYNAME="sp_relay_00_pcc_p_g", FILTERINGMODE=FLOWFILTER;
ADD RULE:RULENAME="sp_relay_00_r_h_003", POLICYTYPE=PCC, FLOWFILTERNAME="sp_relay_00_f_f_h_003", PRIORITY=30282, POLICYNAME="sp_relay_00_pcc_p_g", FILTERINGMODE=FLOWFILTER;
ADD RULE:RULENAME="sp_relay_00_r_h_004", POLICYTYPE=PCC, FLOWFILTERNAME="sp_relay_00_f_f_h_004", PRIORITY=30283, POLICYNAME="sp_relay_00_pcc_p_g", FILTERINGMODE=FLOWFILTER;
ADD RULEBINDING:USERPROFILENAME="testuserprofilename", RULENAME="sp_relay_00_r_h_001", POLICYTYPE=PCC;
ADD RULEBINDING:USERPROFILENAME="testuserprofilename", RULENAME="sp_relay_00_r_h_002", POLICYTYPE=PCC;
ADD RULEBINDING:USERPROFILENAME="testuserprofilename", RULENAME="sp_relay_00_r_h_003", POLICYTYPE=PCC;
ADD RULEBINDING:USERPROFILENAME="testuserprofilename", RULENAME="sp_relay_00_r_h_004", POLICYTYPE=PCC;
```

//执行配置刷新操作，生效媒体中继业务匹配规则。

```
SET REFRESHSRV:REFRESHTYPE=USERPROFILE;
```
