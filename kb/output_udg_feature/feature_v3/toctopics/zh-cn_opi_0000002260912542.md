# 配置RADIUS功能

* 操作场景
* 必备事项
* 任务示例

## 操作场景

该任务旨在指导操作人员完成SMF与AAA(RADIUS Server)的对接配置和功能配置。

## 必备事项

前提条件

* 已完成SMF基础配置。
* 已完成一跳入家要求的移动VPN相关配置。

## 任务示例

任务描述

完成SMF对接RADIUS以及打开RADIUS功能。

脚本

//创建L3VPN实例。

```
ADD L3VPNINST:VRFNAME="vpn_aaa";
```

```
ADD VPNINSTAF: VRFNAME="vpn_aaa", AFTYPE=ipv4uni, VRFLABELMODE=perRoute;
```

//创建VPN实例。

```
ADD VPNINST: VPNINSTANCE ="vpn_aaa";
```

//配置Gi接口。

```
ADD LOGICIP: IPVERSION=IPv4, LOGICIPV4="10.8.20.1", VPNINSTNAME="vpn_aaa";
ADD LOGICINF:NAME="giif1/0/0",IPVERSION=IPV4,IPV4ADDRESS1="10.8.20.1",IPV4MASK1="255.255.255.255",VPNINSTANCE="vpn_aaa";
```

//配置RADIUS服务器组和RADIUS服务器。

```
ADD RDSSVRGRP: RDSSVRGRPNAME="radius_001", MODE=MASTER_SLAVE, ACCTTOAUTH=DISABLE, SIGOPTACCTMSG=DISABLE, ACCTTIMEOUT=10, ACCTRETRANSMIT=5;
```

```
ADD RDSSVR: RDSSVRGRPNAME="radius_001", SERVERTYPE=AUTHENTICATION, IPVERSION=IPV4, SERVERIPV4="10.168.10.1", PORT=1812, VPNINSTANCE="vpn_aaa", CIPHERKEY="*****", CONGESTION=DISABLE, PRIFLAG=PRIMARY, WALVALUE=0, CIPHERKEYCNFM="*****";
```

//配置Gi接口鉴权。

```
ADD RADIUSCLIENTIP: RDSSVRGRPNAME="radius_001", AUTHORACCT=AUTHENTICATION, INTFNAME="giif1/0/0";
```

//配置APN级Gi接口鉴权。如果APN无特别规划，通过**ADD RADIUSCLIENTIP**配置即可。

```
ADD APNRDSCLIENTIP: APN="home.dnn", AUTHORACCT=AUTHENTICATION, INTFNAME="giif1/0/0";
```

//配置对接RADIUS场景时的鉴权能力。

```
SET APNAUTHATTR: APN="home.dnn", ACCESSMODE=NON_TRANS, AUTHMODE=MSISDN, PASSWORD="Pwd001", CFMPASSWORD="Pwd001", COMMONUSERNAME="admin", COMMONUSERPASS="Huawei12#$", CFMCOMMUSERPASS="Huawei12#$";
```

//配置APN绑定RADIUS服务器组。

```
ADD APNRDSSVRGRP:APN="home.dnn", RDSSVRGRPNAME="radius_001";
```

//配置鉴权请求携带的属性。脚本仅为示例，实际能力需与对端AAA服务器协商。

```
SET RDSAUTHREQATTR: RDSSVRGRPNAME="radius_001", ACCTSESSIONID=ENABLE, NASID=ENABLE, NASIDTYPE=APN, IMSI=DISABLE, CHARGINGID=DISABLE, PREPAIDIND=DISABLE, GGSNIP=ENABLE, SGSNIP=ENABLE, REQUESTAPN=DISABLE, GGSNVENDOR=ENABLE, GGSNVERION=ENABLE, RQAUTHTICATOR=DISABLE, ACCTMULSESSID=ENABLE, EVENTTIME=ENABLE, NASPORTIDSW=ENABLE, NASPORTID="UNC", WLANADDR=DISABLE;
```

//配置鉴权请求携带的私有扩展属性。脚本仅为示例，实际能力需与对端AAA服务器协商。

```
SET RDSAUTHREQVSA:RDSSVRGRPNAME="radius_001", THREEGPP=ENABLE, IMSI=ENABLE, CHARGINGID=ENABLE, PDPTYPE=ENABLE, CGADDRESS=ENABLE, NEGOTIAEDQOS=ENABLE, SGSNSGWADDRESS=ENABLE, GGSNADDRESS=ENABLE, IMSIMCCMNC=ENABLE, GGSNMCCMNC=ENABLE, NSAPI=ENABLE, SELECTIONMODE=ENABLE, CHARGINGCHAR=ENABLE, SGSNSGWMCCMNC=ENABLE, IMEISV=ENABLE, RATTYPE=ENABLE, ULI=ENABLE, MSTIMEZONE=ENABLE, NEGOTIAEDDSCP=ENABLE, SGSNIPV6ADDRESS=ENABLE, GGSNIPV6ADDRESS=ENABLE, CGIPV6ADDRESS=ENABLE;
```
