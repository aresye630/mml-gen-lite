# 配置媒体中继业务及UPF选择规则（基于upfSelectRules）

* 操作场景
* 必备事项
* 操作步骤

## 操作场景

本操作指导完成在部署媒体中继业务时，如何配置通过upfSelectRules下发UPF选择规则及业务规则。

## 必备事项

前提条件

媒体中继业务的规则名已全网统一规划。

数据

**图1** PCF配置通过upfSelectRules下发UPF选择规则

## 操作步骤

1. 登录UPCF WebUI界面进行业务配置，在“系统管理 > 批处理 > CLI导入”界面执行以下CLI命令。

   登录UPCF WebUI界面可参考《UPCF 产品文档》>UPCF WebUI GUI参考 > UPCF WebUI > 登录UPCF WebUI界面。
2. 执行如下CLI脚本，创建规则“Rule\_select\_relay\_upf01”，用于当SmfSession.RatType取值为“NR”时，执行动作“UpfSelectRule”，下发UPF预定义规则。

   ```
   /*创建条件组ConditionCG02_relay_5G：当SmfSession.RatType取值为“NR”时，下发规则“Rule_select_relay_upf01”。*/
   ADD CONDITIONGROUP:CONDITIONGROUPNAME="ConditionCG02_relay_5G",RELATION=AND,USETIMEZONE=No;
   ADD CONDITION:CONDITIONGROUPNAME="ConditionCG02_relay_5G",CONDITIONNAME="ConditionCG02_relay_5G",OBJECTATTRIBUTE="SmfSession.RatType",OPERATOR=Equal,RIGHTVALUETYPE=Value,RIGHTVALUE="NR";

   /*创建动作组AG_select_relay_upf：用于下发UPF预定义规则。取值“select_relay_upf”需要和SMF规划保持一致。*/
   ADD NGACTION:ACTIONNAME="AG_select_relay_upf",TYPE="UpfSelectRule";
   ADD NGACTIONATTR:ACTIONNAME="AG_select_relay_upf",ATTRNAME="upfSelectId",TYPE=Value,VALUE="select_relay_upf";
   ADD NGACTIONATTR:ACTIONNAME="AG_select_relay_upf",ATTRNAME="reportInd",TYPE=Value,VALUE="True";

   /*创建规则Rule_select_relay_upf01：绑定条件组ConditionCG02_relay_5G和动作组AG_select_relay_upf，用于控制SmfSession.RatType取值为“NR”时，下发UPF预定义规则。*/
   ADD RULE:RULENAME="Rule_select_relay_upf01",OPERATION=Change Rule,TYPE=Smf Pcc Rule 5G,PRECEDENCE=0,VOLUMEUSAGEROUNDING=No,EDR=OFF,RELATION=AND;
   ADD RULECONDITIONGROUP:RULENAME="Rule_select_relay_upf01",CONDITIONGROUPNAME="ConditionCG02_relay_5G";
   ADD RULENGACTION:RULENAME="Rule_select_relay_upf01",ACTIONNAME="AG_select_relay_upf";
   ```
3. 执行如下CLI脚本，创建规则“Rule\_select\_relay\_upf02”，用于当SmfSession.upfActive取值为“select\_relay\_upf”时，执行动作“PredefinedPccRule”，下发5G预定义规则。

   ```
   /*创建条件组ConditionCG01_relay_5G：当SmfSession.upfActive取值为“select_relay_upf”时，下发规则Rule_select_relay_upf02*/
   ADD CONDITIONGROUP:CONDITIONGROUPNAME="ConditionCG01_relay_5G",RELATION=AND,USETIMEZONE=No;
   ADD CONDITION:CONDITIONGROUPNAME="ConditionCG01_relay_5G",CONDITIONNAME="ConditionCG01_relay_5G",OBJECTATTRIBUTE="SmfSession.upfActive",OPERATOR=Equal,RIGHTVALUETYPE=Value,RIGHTVALUE="select_relay_upf";

   /*创建动作组AG_relayrule：用于下发5G预定义规则。取值“testuserprofilename”需要和SMF规划保持一致。*/
   ADD NGACTION:ACTIONNAME="AG_relayrule",TYPE="PredefinedPccRule";
   ADD NGACTIONATTR:ACTIONNAME="AG_relayrule",ATTRNAME="pccRuleId",TYPE=Value,VALUE="testuserprofilename";

   /*创建规则Rule_select_relay_upf02：绑定条件组ConditionCG01_relay_5G和动作组AG_relayrule，用于控制SmfSession.upfActive取值为“select_relay_upf”时，下发5G预定义规则。*/
   ADD RULE:RULENAME="Rule_select_relay_upf02",OPERATION=Change Rule,TYPE=Smf Pcc Rule 5G,PRECEDENCE=0,VOLUMEUSAGEROUNDING=No,EDR=OFF,RELATION=AND;
   ADD RULECONDITIONGROUP:RULENAME="Rule_select_relay_upf02",CONDITIONGROUPNAME="ConditionCG01_relay_5G";
   ADD RULENGACTION:RULENAME="Rule_select_relay_upf02",ACTIONNAME="AG_relayrule";
   ```
4. 添加策略“relay\_policy”，策略类型为N7 Policy，在用户上线时下发，配置触发器“IPCAN\_EST”和“RAT\_TY\_CH”，同时关联规则“Rule\_select\_relay\_upf01”和“Rule\_select\_relay\_upf02”。

   ```
   /*创建策略relay_policy：绑定规则Rule_select_relay_upf01和Rule_select_relay_upf02，触发器为IPCAN_EST和RAT_TY_CH*/
   ADD POLICY:POLICYNAME="relay_policy",TYPE=N7 Policy;
   ADD POLICYRULE:POLICYNAME="relay_policy",RULENAME="Rule_select_relay_upf01";
   ADD POLICYRULE:POLICYNAME="relay_policy",RULENAME="Rule_select_relay_upf02";
   ADD POLICYTRIGGER:POLICYNAME="relay_policy",TRIGGERNAME="IPCAN_EST";
   ADD POLICYTRIGGER:POLICYNAME="relay_policy",TRIGGERNAME="RAT_TY_CH";
   ```
5. 添加业务“service\_relay”，业务类型为VALUE\_ADDED\_SERVICE，用于关联策略“relay\_policy”。

   ```
   /*创建业务service_relay：绑定策略relay_policy。*/
   ADD SERVICE:SERVICENAME="service_relay",SERVICETYPE=VALUE_ADDED_SERVICE,ACTIVATEDBY=PCEF,PRECEDENCE=0,QOSMODE=Replace,SUBSCRIPTIONFORCED=No,PROBILLDATE=No,QUOTAALLOCATIONMODE=None,ISSETSHAREDMODE=No,RESUBSCRIPTIONOPERATION=NULL,ISGROUPSERVICE=No,ISTEMPLATE=No,ISRECUSRLOC=No,APNMATCHTYPE=Full Matching,HASBASESERVICE=No,ADDONLEVEL=Exhaust,CALCVALIDFORDYNSUBGROUP=No,CUSTOMIZEDRESET=No,MASTER2MASTERSHARESERVICE=No;
   ADD SERVICEPOLICY:SERVICENAME="service_relay",POLICYNAME="relay_policy";
   ```
6. 发放业务。
   1. 执行**ADD PSUB**增加一个用户。

      ```
      ADD PSUB: USRIDENTIFIER="xxxxx", USRSTATUS=Custom1, USRBILLINGTYPE=Postpaid, USRGRADE=NULL, USRBILLCYCLEDATE=5, ESSTATUS=NotEmergency, PACKAGETYPE=NULL;
      ```
   2. 执行**ADD PSRV**给用户签约业务。

      ```
      ADD PSRV: USRIDENTIFIER="xxxxx", SRVNAME="service_relay", SRVUSAGESTATE=Normal, SRVROAMINGTYPE=NULL, SRVCONTACTMETHOD=None, SRVCREATESUBSCRIBER=No, PAYMENTFLAG=Yes, SRVEXATTR1=255
      ```
