# 配置DNS代理规则

- 操作场景
- 必备事项
- 任务示例

## 操作场景

媒体中继业务中，UPF需要识别DNS请求中是否携带OTT规划的relay域名来作DNS代理，此时需要在UPF上配置对应的URL识别规则。

## 任务示例

任务描述

基于规划的点播域名**relay.spvod.com**，直播域名**relay.spliving.com**配置7层识别规则，绑定到媒体中继签约套餐对应的预定义规则**testuserprofilename**。

脚本

//配置DNS代理的流过滤器。

```
ADD FILTER:FILTERNAME="relay_f_dns", L34PROTTYPE=STRING, L34PROTOCOL=UDP, MSSTARTPORT=0, MSENDPORT=65535, SVRSTARTPORT=53, SVRENDPORT=53;
ADD FILTERIPV6: FILTERNAME="ipv6_relay_f_dns", L34PROTTYPEV6=STRING, L34PROTOCOL=UDP, SVRSTARTPORT=53, SVRENDPORT=53;
ADD L7FILTER:L7FILTERNAME="sp_relay_d_00_l7_f_001", SUBL7FLTNAME="sp_relay_d_00_sl7_f_100", URLTYPE=URL, URL="relay.spliving.com", METHODTYPE=CONNECT-1&DELETE-1&GET-1&HEAD-1&OPTIONS-1&POST-1&PUT-1&TRACE-1, ISREFEREREN=DISABLE, MMSBLURMATCH=DISABLE;
ADD L7FILTER:L7FILTERNAME="sp_relay_d_00_l7_f_100", SUBL7FLTNAME="sp_relay_d_00_sl7_f_101", URLTYPE=URL, URL="relay.spvod.com", METHODTYPE=CONNECT-1&DELETE-1&GET-1&HEAD-1&OPTIONS-1&POST-1&PUT-1&TRACE-1, ISREFEREREN=DISABLE, MMSBLURMATCH=DISABLE;
```

//增加域名DNS代理规则流过滤器。

```
ADD FLOWFILTER: FLOWFILTERNAME="sp_relay_d_00_f_f_h_001", TETHERDETFLAG=NONE, FLOWFILTERTYPE=USERCONFIG;
ADD FLOWFILTER: FLOWFILTERNAME="sp_relay_d_00_f_f_h_002", TETHERDETFLAG=NONE, FLOWFILTERTYPE=USERCONFIG;
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_d_00_f_f_h_001", FILTERNAME="relay_f_dns";
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_d_00_f_f_h_001", FILTERNAME="ipv6_relay_f_dns";
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_d_00_f_f_h_002", FILTERNAME="relay_f_dns";
ADD FLTBINDFLOWF:FLOWFILTERNAME="sp_relay_d_00_f_f_h_002", FILTERNAME="ipv6_relay_f_dns";
ADD PROTBINDFLOWF: FLOWFILTERNAME="sp_relay_d_00_f_f_h_001", PROTOCOLNAME="dns", L7FILTERNAME="sp_relay_d_00_l7_f_001";
ADD PROTBINDFLOWF: FLOWFILTERNAME="sp_relay_d_00_f_f_h_002", PROTOCOLNAME="dns", L7FILTERNAME="sp_relay_d_00_l7_f_100";
```

//配置DNS代理规则并绑定到媒体中继签约套餐。

```
ADD RULE: RULENAME="sp_relay_d_00_r_h_001", POLICYTYPE=WORKER, FILTERINGMODE=FLOWFILTER, FLOWFILTERNAME="sp_relay_d_00_f_f_h_001", WORKERNAME="relay_dns", WORKERPARA="sp_relay_tpl";
ADD RULE: RULENAME="sp_relay_d_00_r_h_002", POLICYTYPE=WORKER, FILTERINGMODE=FLOWFILTER, FLOWFILTERNAME="sp_relay_d_00_f_f_h_002", WORKERNAME="relay_dns", WORKERPARA="sp_relay_tpl";
ADD RULEBINDING: USERPROFILENAME="testuserprofilename", RULENAME="sp_relay_d_00_r_h_001", POLICYTYPE=WORKER;
ADD RULEBINDING: USERPROFILENAME="testuserprofilename", RULENAME="sp_relay_d_00_r_h_002", POLICYTYPE=WORKER;
```

//执行配置刷新操作，生效媒体中继DNS代理规则

```
SET REFRESHSRV:REFRESHTYPE=USERPROFILE;
```
